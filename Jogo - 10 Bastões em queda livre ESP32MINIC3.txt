/*
=========================================================
 JOGO DOS BASTÕES - VERSÃO REVISADA E ESTÁVEL
 ESP32-C3 SUPER MINI | Arduino IDE
 
 - 10 Relés para Bastões (ativos em LOW)
 - 1 Relé para LED Verde 12V na GPIO 8 (ativo em LOW)
 - LED Interno Desativado para liberar GPIO 8
=========================================================
*/

// ------------ CONFIGURAÇÃO DOS PINOS ------------

const byte PINO_BOTAO   = 2; // Botão conectado na GPIO 2 e no GND
const byte PINO_RELE_LED = 8; // Relé do LED Verde conectado na GPIO 8 (Antigo LED interno)

// 10 pinos digitais seguros para os bastões
const byte rele[] = {
  0,   // Relé 1 (GPIO 0)
  1,   // Relé 2 (GPIO 1)
  3,   // Relé 3 (GPIO 3)
  4,   // Relé 4 (GPIO 4)
  5,   // Relé 5 (GPIO 5)
  6,   // Relé 6 (GPIO 6)
  7,   // Relé 7 (GPIO 7)
  10,  // Relé 8 (GPIO 10)
  20,  // Relé 9 (GPIO 20 / RX)
  21   // Relé 10 (GPIO 21 / TX)
};

const byte TOTAL_RELES = sizeof(rele) / sizeof(rele[0]); 

// ------------ TEMPOS ------------

const unsigned long TEMPO_RELE = 2000;       // ms ligado
const int ESPERA_MIN = 0;                    // segundos
const int ESPERA_MAX = 4;                    // segundos

// ------------ ESTADOS DO JOGO ------------

enum EstadoJogo {
  AGUARDANDO_START,
  CONTAGEM_REGRESSIVA,
  BASTAO_LIGADO,
  BASTAO_ESPERA,
  FIM_JOGO
};

EstadoJogo estadoAtual = AGUARDANDO_START;

// ------------ VARIÁVEIS DE CONTROLE ------------

byte ordem[10]; 
byte indiceBastao = 0;

unsigned long tempoAnteriorEstado = 0;
unsigned long tempoEsperaAtual = 0;

bool botaoPressionadoAnterior = false;
unsigned long tempoDebounceBotao = 0;
int ultimoSegundoPrintado = 0;

// ------------ FUNÇÃO AUXILIAR ------------
void activarBastao(unsigned long tempoAtual) {
  byte indiceSorteado = ordem[indiceBastao];
  byte pinoFisico = rele[indiceSorteado];

  Serial.print("Bastao ");
  Serial.print(indiceSorteado + 1);
  Serial.print(" (GPIO ");
  Serial.print(pinoFisico);
  Serial.println(") -> CAINDO!");

  digitalWrite(pinoFisico, LOW); // Liga o relé do bastão (ativo em LOW)
  tempoAnteriorEstado = tempoAtual;
  estadoAtual = BASTAO_LIGADO;
}

// ------------ SETUP ------------
void setup()
{
  Serial.begin(115200);

  pinMode(PINO_BOTAO, INPUT_PULLUP);
  
  // Inicializa o relé do LED de 12V (Inicia ACESO -> LOW)
  pinMode(PINO_RELE_LED, OUTPUT);
  digitalWrite(PINO_RELE_LED, LOW); 

  // Inicializa todos os relés dos bastões desligados (HIGH)
  for (byte i = 0; i < TOTAL_RELES; i++)
  {
    pinMode(rele[i], OUTPUT);
    digitalWrite(rele[i], HIGH);   
  }

  randomSeed(esp_random());

  Serial.println("\n=================================");
  Serial.println(" JOGO DOS BASTOES - ESP32-C3");
  Serial.print(" Configurado para: ");
  Serial.print(TOTAL_RELES);
  Serial.println(" Bastões");
  Serial.println(" Aguardando START...");
  Serial.println("=================================");
}

// ------------ LOOP ------------
void loop()
{
  unsigned long tempoAtual = millis();

  // --- MÁQUINA DE ESTADOS DO JOGO ---
  switch (estadoAtual)
  {
    case AGUARDANDO_START:
      digitalWrite(PINO_RELE_LED, LOW); // LED Verde 12V ACESO fixo (Aguardando start)
      
      if (digitalRead(PINO_BOTAO) == LOW && !botaoPressionadoAnterior)
      {
        if (tempoAtual - tempoDebounceBotao > 50) 
        {
          botaoPressionadoAnterior = true;
          tempoAnteriorEstado = tempoAtual;
          estadoAtual = CONTAGEM_REGRESSIVA;
          ultimoSegundoPrintado = 0;
          Serial.println("\nSTART");
        }
      }
      if (digitalRead(PINO_BOTAO) == HIGH)
      {
        botaoPressionadoAnterior = false;
        tempoDebounceBotao = tempoAtual;
      }
      break;

    case CONTAGEM_REGRESSIVA:
      {
        unsigned long tempoDecorrido = tempoAtual - tempoAnteriorEstado;

        // Controle dos segundos no Monitor Serial
        if (tempoDecorrido < 1000 && ultimoSegundoPrintado == 0) {
          Serial.println("3");
          ultimoSegundoPrintado = 1;
        }
        else if (tempoDecorrido >= 1000 && tempoDecorrido < 2000 && ultimoSegundoPrintado == 1) {
          Serial.println("2");
          ultimoSegundoPrintado = 2;
        }
        else if (tempoDecorrido >= 2000 && tempoDecorrido < 3000 && ultimoSegundoPrintado == 2) {
          Serial.println("1");
          ultimoSegundoPrintado = 3;
        }

        // --- PISCA LED 12V (500ms ON / 200ms OFF) ---
        // Lógica invertida do relé: LOW = ON (Aceso), HIGH = OFF (Apagado)
        if (tempoDecorrido % 700 < 500) {
          digitalWrite(PINO_RELE_LED, LOW);  // LIGA o relé -> LIGA o LED
        } else {
          digitalWrite(PINO_RELE_LED, HIGH); // DESLIGA o relé -> DESLIGA o LED
        }

        if (tempoDecorrido >= 3000) {
          Serial.println("VALENDO!");
          
          digitalWrite(PINO_RELE_LED, HIGH); // DESLIGA o LED 12V durante o jogo ativo
          
          // Inicializa e embaralha os bastões
          for (byte i = 0; i < TOTAL_RELES; i++) ordem[i] = i;
          for (int i = TOTAL_RELES - 1; i > 0; i--)
          {
            byte j = random(0, i + 1);
            byte temp = ordem[i];
            ordem[i] = ordem[j];
            ordem[j] = temp;
          }

          indiceBastao = 0;
          activarBastao(tempoAtual); 
        }
      }
      break;

    case BASTAO_LIGADO:
      digitalWrite(PINO_RELE_LED, HIGH); // Garante LED 12V apagado durante o jogo

      if (tempoAtual - tempoAnteriorEstado >= TEMPO_RELE)
      {
        digitalWrite(rele[ordem[indiceBastao]], HIGH); // Desliga o relé do bastão atual

        if (indiceBastao >= TOTAL_RELES - 1)
        {
          estadoAtual = FIM_JOGO;
          tempoAnteriorEstado = tempoAtual;
        }
        else
        {
          tempoEsperaAtual = random(ESPERA_MIN, ESPERA_MAX + 1) * 1000UL; 
          Serial.print("Aguardando ");
          Serial.print(tempoEsperaAtual / 1000UL);
          Serial.println(" segundos...");
          
          tempoAnteriorEstado = tempoAtual;
          estadoAtual = BASTAO_ESPERA;
        }
      }
      break;

    case BASTAO_ESPERA:
      digitalWrite(PINO_RELE_LED, HIGH); // Garante LED 12V apagado durante o jogo

      if (tempoAtual - tempoAnteriorEstado >= tempoEsperaAtual)
      {
        indiceBastao++;
        activarBastao(tempoAtual); 
      }
      break;

    case FIM_JOGO:
      digitalWrite(PINO_RELE_LED, LOW); // Acende fixo o LED 12V indicando prontidão
      
      Serial.println();
      Serial.println("=================================");
      Serial.println(" FIM DO JOGO!");
      Serial.println(" Aguardando novo jogo...");
      Serial.println("=================================");
      estadoAtual = AGUARDANDO_START;
      break;
  }
}
